The link you clicked on isn't here — but that's okay, man. That's okay.

<script>
  // Suppress GoatCounter's automatic pageview — we fire a custom event instead
  // so 404 hits appear in the events list and don't pollute page traffic stats.
  window.goatcounter = { no_onload: true };

  // Jekyll is static, so it can't know the bad URL at build time.
  // This script runs immediately when the page loads and appends the actual
  // requested URL (in monospace <code>) to the end of the preceding paragraph,
  // giving the visitor something useful to read/copy.
  (function () {
    var url = document.createElement("code");
    url.textContent = window.location.href;
    var span = document.createElement("span");
    span.appendChild(document.createTextNode(" ("));
    span.appendChild(url);
    span.appendChild(document.createTextNode(")"));
    document.currentScript.previousElementSibling.appendChild(span);
  })();

  // After GoatCounter's count.js has loaded, fire a single event recording
  // the bad URL so broken links can be spotted and fixed.
  window.addEventListener('load', function () {
    if (window.goatcounter && window.goatcounter.count) {
      window.goatcounter.count({
        path: '404: ' + window.location.pathname,
        title: '404 Not Found: ' + window.location.href,
        event: true
      });
    }
  });
</script>
